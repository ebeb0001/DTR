\begin{lstlisting}[language=Python, , basicstyle=\ttfamily, frame=single, caption={data extraction code}]
import requests
import json

def get_query(area : int, filter : str) :
    query : str = f"""
    [out:json]
    [timeout:60]
    ;
    area({area})->.searchArea;
    (
    nwr
        {filter}
        (area.searchArea);
    );
    out geom;
    """
    return query

def export_railway_network(area : int) :
    URL = "https://overpass-api.de/api/interpreter"
    filters = {
        "track_sections" : "['railway'='rail']",
        "buffer_stops" : "['railway'='buffer_stop']",
        "switches" : "['railway'='switch']",
        "signals" : "['railway'='signal']",
        "stops" : "['railway'='stop']",
        "stations" : "['railway'='station']['operator'='NMBS/SNCB']",
    }
    data = {}
    for filter in filters :
        query = get_query(area, filters[filter])
        response = requests.get(URL, data=query)
        if response.status_code == 200 :
            data[filter] = response.json()
        response.close()
    return data

if __name__ == "__main__" :
    data = export_railway_network(3600386970)
    for name in data :
        with open(f"data/{name}.json", "w", encoding="utf-8") as f :
            json.dump(data[name], f, indent=4, ensure_ascii=False)
\end{lstlisting}
}

                t_coords = get_coords(t)
                if coord in t_coords :
                    index = track_coords.index(coord)
                    if index != 0 and index != (len(track_coords) - 1) :
                        sep.add(index)
    sep = list(sep)
    sep.sort()
    tracks_sep[track["id"]] = 
    [sep if len(sep) > 0 else None, f"size : {len(track_coords)}"]
    
track_sections_updated = {}
for track in track_sections :
    cmpt = 0
    track_id = track["id"]
    sep = tracks_sep[track_id][0]
    track_coords = get_coords(track)
    track_sections_updated[track_id] = []
    separators = tracks_sep[track_id][0]
    if separators is None :
        track_sections_updated[track_id] = track_coords
    else :
        for sep in separators :
            track_sections_updated[track_id].append([track_coords[cmpt]])
            while (cmpt + 1) <= sep and (cmpt + 1) < len(track_coords) :
                cmpt += 1
                track_sections_updated[track_id][-1].append(track_coords[cmpt])
        track_sections_updated[track_id].append([track_coords[cmpt]])
        for i in range(cmpt + 1, len(track_coords)) :
            track_sections_updated[track_id][-1].append(track_coords[i])
